# Python program to calculate the first 10,000 terms of the series and multiply by 4

total = 0
for i in range(10000):
    term = 1 / (2 * i + 1)
    if i % 2 == 0:
        total += term  # Add for even indices
    else:
        total -= term  # Subtract for odd indices

# Multiply total by 4
result = total * 4

print(result)
